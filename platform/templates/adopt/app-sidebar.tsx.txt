"use client";

import { useTranslations } from "next-intl";
import { Link, useRouter } from "@web-app-starter/i18n/navigation";
import { appConfig } from "@web-app-starter/app-config";
import { OrganizationSwitcher, useAccountNavigation, useOrganizations, useSignOut } from "@web-app-starter/auth-ui";
import {
  Sidebar, SidebarContent, SidebarFooter, SidebarHeader,
  SidebarMenu, SidebarMenuButton, SidebarMenuItem,
} from "@web-app-starter/design-system";
import { AppLogo } from "@/components/app-logo";

export function AppSidebar({ displayName, displayEmail, onNavigateHome }: {
  displayName: string;
  displayEmail?: string;
  onNavigateHome?: () => void;
}) {
  const tc = useTranslations("common");
  const td = useTranslations("dashboard");
  const to = useTranslations("organization");
  const organizations = useOrganizations();
  const accountNavigation = useAccountNavigation();
  const router = useRouter();
  const signOut = useSignOut("/");
  return (
    <Sidebar>
      <SidebarHeader>
        <OrganizationSwitcher logo={<AppLogo size={20} />} productName={appConfig.identity.productName} />
      </SidebarHeader>
      <SidebarContent>
        <SidebarMenu>
          <SidebarMenuItem>
            <SidebarMenuButton onClick={onNavigateHome ?? (() => router.push("/dashboard"))}>
              {tc("backToHome")}
            </SidebarMenuButton>
          </SidebarMenuItem>
          <SidebarMenuItem>
            <SidebarMenuButton asChild><Link href={accountNavigation.settingsHref}>{td("account")}</Link></SidebarMenuButton>
          </SidebarMenuItem>
          {organizations?.activeOrganizationId && <SidebarMenuItem>
            <SidebarMenuButton asChild><Link href={accountNavigation.organizationHref}>{to("label")}</Link></SidebarMenuButton>
          </SidebarMenuItem>}
        </SidebarMenu>
      </SidebarContent>
      <SidebarFooter className="p-4">
        <span className="truncate">{displayName}</span>
        <span className="truncate text-sm text-muted-foreground">{displayEmail}</span>
        <SidebarMenuButton onClick={signOut}>{tc("signOut")}</SidebarMenuButton>
      </SidebarFooter>
    </Sidebar>
  );
}
